cmake_minimum_required(VERSION 3.27)
project(windows_rt_diagnostic LANGUAGES CXX)
if(NOT DEFINED ORPHEUS_ROOT OR NOT DEFINED DIAG_OUTPUT)
  message(FATAL_ERROR "ORPHEUS_ROOT and DIAG_OUTPUT are required")
endif()
set(ORPHEUS_ENABLE_REALTIME ON CACHE BOOL "" FORCE)
set(ORP_ENABLE_ASAN OFF CACHE BOOL "" FORCE)
set(ORP_ENABLE_UBSAN OFF CACHE BOOL "" FORCE)
set(ORP_WITH_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${ORPHEUS_ROOT}" "${CMAKE_BINARY_DIR}/orpheus" EXCLUDE_FROM_ALL)
add_executable(windows_rt_diagnostic "${ORPHEUS_ROOT}/tools/windows_rt_diagnostic.cpp")
target_compile_features(windows_rt_diagnostic PRIVATE cxx_std_20)
target_include_directories(windows_rt_diagnostic PRIVATE "${DIAG_OUTPUT}" "${ORPHEUS_ROOT}/include" "${ORPHEUS_ROOT}/src/core")
target_link_libraries(windows_rt_diagnostic PRIVATE orpheus_routing)
if(MSVC)
  set_source_files_properties("${ORPHEUS_ROOT}/src/core/routing/routing_matrix.cpp" TARGET_DIRECTORY orpheus_routing PROPERTIES
    COMPILE_OPTIONS "/FAcs;/Fa${DIAG_OUTPUT}/routing_matrix.asm")
endif()
get_target_property(_defs orpheus_routing COMPILE_DEFINITIONS)
get_target_property(_incs orpheus_routing INCLUDE_DIRECTORIES)
get_target_property(_opts orpheus_routing COMPILE_OPTIONS)
message(STATUS "DIAGNOSTIC routing compile definitions: ${_defs}")
message(STATUS "DIAGNOSTIC routing include directories: ${_incs}")
message(STATUS "DIAGNOSTIC routing compile options: ${_opts}")
message(STATUS "DIAGNOSTIC routing assembly: ${DIAG_OUTPUT}/routing_matrix.asm")
