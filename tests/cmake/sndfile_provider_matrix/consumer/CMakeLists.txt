cmake_minimum_required(VERSION 3.22)
project(sndfile_provider_consumer LANGUAGES CXX)

if(NOT DEFINED ORPHEUS_EXPECT_PROVIDER)
  message(FATAL_ERROR "ORPHEUS_EXPECT_PROVIDER is required")
endif()

find_package(OrpheusSDK CONFIG REQUIRED)

get_target_property(_links Orpheus::orpheus_audio_utils INTERFACE_LINK_LIBRARIES)
get_target_property(_link_dirs Orpheus::orpheus_audio_utils INTERFACE_LINK_DIRECTORIES)
if(_links STREQUAL "_links-NOTFOUND")
  set(_links "")
endif()
if(_link_dirs AND NOT _link_dirs STREQUAL "_link_dirs-NOTFOUND")
  message(FATAL_ERROR "exported audio utils exposes link directories: ${_link_dirs}")
endif()

if(ORPHEUS_EXPECT_PROVIDER STREQUAL "SndFile")
  if(NOT TARGET SndFile::sndfile OR NOT _links MATCHES "(^|;)SndFile::sndfile(;|$)")
    message(FATAL_ERROR "SndFile provider was not resolved exactly: ${_links}")
  endif()
elseif(ORPHEUS_EXPECT_PROVIDER STREQUAL "PkgConfig")
  if(NOT TARGET PkgConfig::SNDFILE OR NOT _links MATCHES "(^|;)PkgConfig::SNDFILE(;|$)")
    message(FATAL_ERROR "PkgConfig provider was not resolved exactly: ${_links}")
  endif()
elseif(ORPHEUS_EXPECT_PROVIDER STREQUAL "None")
  if(_links MATCHES "sndfile|SndFile|SNDFILE")
    message(FATAL_ERROR "None provider still exports sndfile: ${_links}")
  endif()
else()
  message(FATAL_ERROR "Unknown provider expectation: ${ORPHEUS_EXPECT_PROVIDER}")
endif()

add_executable(sndfile_provider_consumer main.cpp)
target_compile_features(sndfile_provider_consumer PRIVATE cxx_std_20)
target_compile_definitions(sndfile_provider_consumer PRIVATE
  ORPHEUS_EXPECT_PROVIDER="${ORPHEUS_EXPECT_PROVIDER}")
target_link_libraries(sndfile_provider_consumer PRIVATE Orpheus::audio_io)
